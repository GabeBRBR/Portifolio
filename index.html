Automações, Scripts & Algoritmos

# -*- coding: utf-8 -*-
"""
AUTOMAÇÃO BIM: Parametrização em Lote e Extração de Quantitativos no Revit
Autor: Gabriel Biagini Almeida Reis
Ambiente: Autodesk Revit / Dynamo Python Node (pyRevit)
"""
import clr
clr.AddReference('RevitAPI')
clr.AddReference('RevitServices')

from Autodesk.Revit.DB import *
from RevitServices.Persistence import DocumentManager
from RevitServices.Transactions import TransactionManager

doc = DocumentManager.Instance.CurrentDBDocument

def executar_parametrizacao_mep():
    # Coleta todas as curvas de tubulação do projeto
    collector = FilteredElementCollector(doc)\
        .OfCategory(BuiltInCategory.OST_PipeCurves)\
        .WhereElementIsNotElementType()\
        .ToElements()

    quantitativos = []
    TransactionManager.Instance.EnsureInTransaction(doc)
    
    for pipe in collector:
        d_param = pipe.LookupParameter("Diâmetro")
        l_param = pipe.LookupParameter("Comprimento")
        s_param = pipe.LookupParameter("Tipo de Sistema")
        
        if d_param and l_param and s_param:
            diam_mm = round(d_param.AsDouble() * 304.8, 1)   # Pés para mm
            comp_m = round(l_param.AsDouble() * 0.3048, 2)   # Pés para metros
            sistema = s_param.AsString()
            
            # Gera Tag Rastreável Padronizada
            codigo_tag = f"MEP-{sistema[:3].upper()}-DN{int(diam_mm)}"
            param_rastreio = pipe.LookupParameter("Código_Rastreabilidade")
            if param_rastreio and not param_rastreio.IsReadOnly:
                param_rastreio.Set(codigo_tag)
                
            quantitativos.append({"ID": pipe.Id.IntegerValue, "Sistema": sistema, "DN_mm": diam_mm, "L_m": comp_m})
            
    TransactionManager.Instance.TransactionTaskDone()
    return f"Sucesso: {len(quantitativos)} elementos MEP parametrizados e quantificados."

OUT = executar_parametrizacao_mep()

"""
TCC - UFG | Calibração do Modelo Concrete Damaged Plasticity (CDP) no ABAQUS
Pesquisa: Vigas de Concreto Armado com Agregado Reciclado
Autor: Gabriel Biagini Almeida Reis
"""
import numpy as np

def calibrar_propriedades_cdp(fck, taxa_reciclado=0.30):
    # Ajuste do módulo de elasticidade com base no percentual de agregado reciclado
    E_convencional = 5600.0 * np.sqrt(fck) # MPa
    E_ajustado = E_convencional * (1.0 - (0.15 * taxa_reciclado))
    
    # Parâmetros constitutivos do ABAQUS CDP
    cdp_plasticity = {
        "dilation_angle": 36.0,
        "eccentricity": 0.1,
        "fb0_fc0": 1.16,
        "k_shape": 0.667,
        "viscosity": 0.0005
    }
    
    # Geração dos pontos da curva de Compressão Inelástica
    deformacoes = np.linspace(0.0002, 0.0035, 12)
    tensoes = fck * (2*(deformacoes/0.002) - (deformacoes/0.002)**2)
    def_inelastica = deformacoes - (tensoes / E_ajustado)
    
    # Evolução do dano à compressão (dc)
    dano_c = np.clip(1.0 - (tensoes / (fck * 1.05)), 0.0, 0.95)
    
    return cdp_plasticity, list(zip(tensoes, def_inelastica)), list(zip(dano_c, def_inelastica))

params, curva_tensao, curva_dano = calibrar_propriedades_cdp(fck=30.0, taxa_reciclado=0.50)

"""
PROCESSAMENTO DE NUVEM DE PONTOS (As-Built Laser Scanning)
Segmentação de planos arquitetônicos e filtragem estatística
Autor: Gabriel Biagini Almeida Reis
"""
import open3d as o3d

def tratar_nuvem_para_revit(arquivo_e57_ou_ply):
    pcd = o3d.io.read_point_cloud(arquivo_e57_ou_ply)
    
    # 1. Filtro Estatístico de Ruídos (Statistical Outlier Removal)
    pcd_limpa, _ = pcd.remove_statistical_outlier(nb_neighbors=20, std_ratio=2.0)
    
    # 2. Downsampling Voxel para otimização de performance no BIM
    pcd_down = pcd_limpa.voxel_down_sample(voxel_size=0.015) # Grid 15mm
    
    # 3. RANSAC para detecção automática de planos de parede e pisos
    plane_model, inliers = pcd_down.segment_plane(distance_threshold=0.01, ransac_n=3, num_iterations=1000)
    a, b, c, d = plane_model
    print(f"Plano Estrutural Detectado: {a:.3f}x + {b:.3f}y + {c:.3f}z + {d:.3f} = 0")
    
    o3d.io.write_point_cloud("asbuilt_otimizado.pts", pcd_down)
    return "Nuvem de pontos tratada com sucesso."

"""
AUDITORIA AUTOMATIZADA DE PROJETOS DE ENGENHARIA CIVIL COM IA
Validação de critérios de durabilidade e cobrimento conforme ABNT NBR 6118:2023
Autor: Gabriel Biagini Almeida Reis
"""
import json

def auditar_durabilidade_nbr6118(dados_viga):
    cobrimento_minimo = { "I": 20, "II": 25, "III": 35, "IV": 45 } # mm
    classe = dados_viga.get("classe_agressividade", "II")
    adotado = dados_viga.get("cobrimento_mm", 20)
    minimo = cobrimento_minimo.get(classe, 25)
    
    resultado = {
        "elemento": dados_viga.get("nome", "Viga V101"),
        "classe": classe,
        "cobrimento_adotado_mm": adotado,
        "cobrimento_minimo_mm": minimo,
        "conforme": adotado >= minimo,
        "parecer": "Aprovado" if adotado >= minimo else f"Reprovado: Adotar mínimo de {minimo}mm (NBR 6118 Tab 7.2)"
    }
    return json.dumps(resultado, indent=2, ensure_ascii=False)
